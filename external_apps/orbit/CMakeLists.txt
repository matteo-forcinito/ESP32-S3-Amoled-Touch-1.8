cmake_minimum_required(VERSION 3.16)
if(DEFINED ENV{WATCH_OS_COMMON_DIR})
    set(WATCH_OS_COMMON "$ENV{WATCH_OS_COMMON_DIR}")
elseif(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../watch-os-common/components")
    set(WATCH_OS_COMMON "${CMAKE_CURRENT_LIST_DIR}/../../watch-os-common")
else()
    set(WATCH_OS_COMMON "${CMAKE_CURRENT_LIST_DIR}/../../../watch-os-common")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../components" "${WATCH_OS_COMMON}/components")
set(EXCLUDE_COMPONENTS "apps" "companion" "webradio" "hid_link" "deck_core" "watcher_camera" "bt_speaker")
if(DEFINED ENV{ORBIT_VERSION})
    set(PROJECT_VER "$ENV{ORBIT_VERSION}")
else()
    set(PROJECT_VER "2.0.0")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
# -O2 (sdkconfig.defaults) finds more truncation/overflow cases in shared OS code: warn, do not fail.
idf_build_set_property(COMPILE_OPTIONS "-Wno-error=format-truncation;-Wno-error=stringop-truncation;-Wno-error=stringop-overflow;-Wno-error=array-bounds;-Wno-error=maybe-uninitialized" APPEND)
project(orbit)
